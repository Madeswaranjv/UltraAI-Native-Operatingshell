# ║  Merges the best of four schema lineages:                                   ║

# ║  Philosophy: Describe WHAT and WHY. Never HOW.                              ║

# ═══════════════════════════════════════════════════════════════════════════════
# SECTION 1 · MANIFEST
# Identity, classification, and human-readable intent for the system.
# This is the first thing an LLM reads — make it unambiguous.
# ═══════════════════════════════════════════════════════════════════════════════

manifest:
  id: <unique-system-slug>            # machine-safe, kebab-case  e.g. "sentinel-fraud-v2"
  name: <Human Readable System Name>
  domain: <banking|ai|blockchain|saas|healthcare|logistics|...>
  version: "<semver>"                 # system version, not schema version
  intent: >
    One or two sentences describing what this system exists to do.
    Written in plain language. This is consumed by Ultra before any planning.
  owner: <team-or-org-slug>
  stability: draft | stable | frozen
    # draft   = architecture is evolving; Ultra may replan freely
    # stable  = architecture is validated; Ultra should not invent new structure
    # frozen  = no structural changes permitted; Ultra treats this as read-only spec
  criticality: mission-critical | business-critical | standard | experimental
  tags: []
  metadata: {}                        # extension point — any custom key-value pairs


# ═══════════════════════════════════════════════════════════════════════════════
# SECTION 2 · SYSTEM PROPERTIES
# Cross-cutting, system-wide behavioral contracts.
# NEW in v2.0 (from S2): consistency_model, failure_mode, observability_level.
# Ultra uses these when planning how to handle uncertainty or partial failure.
# ═══════════════════════════════════════════════════════════════════════════════

properties:
  consistency_model: strong | eventual | causal | session
    # strong:   all reads reflect the latest write
    # eventual: system converges but may serve stale reads temporarily
    # causal:   causally related operations are seen in order
    # session:  consistency guaranteed within a user session

  failure_mode: fail-stop | fail-safe | degrade-gracefully
    # fail-stop:          halt all execution on any critical failure
    # fail-safe:          default to a known-safe state on failure
    # degrade-gracefully: shed non-critical capability; preserve core path

  observability_level: full | metrics-only | logs-only | minimal
    # full:         traces, metrics, logs, and audit records
    # metrics-only: quantitative signals only
    # logs-only:    event records only
    # minimal:      only errors and fatal conditions

  security_model: "<zero-trust|defense-in-depth|perimeter|...>"
    # describe the overarching security philosophy in one phrase


# ═══════════════════════════════════════════════════════════════════════════════
# SECTION 3 · AXIOMS
# System-wide truths that MUST hold at all times.
# Ultra checks axioms BEFORE and AFTER every stratum execution.
# A violated hard axiom halts execution; a soft axiom emits a warning.
#
# NEW in v2.0: constraint_type field now includes "ethical" as a first-class
# category alongside technical, operational, resource, and compliance types.
# ═══════════════════════════════════════════════════════════════════════════════

axioms:
  - id: AX-<NNN>                      # e.g. AX-001
    statement: >
      A declarative truth, not a rule. Frame as a fact that must remain true,
      not as an instruction. E.g. "Every decision is traceable to its inputs."
    constraint_type: technical | operational | ethical | compliance | resource
      # technical:   architectural or system integrity truth
      # operational: runtime behavior truth
      # ethical:     value alignment and harm-avoidance truth  ← NEW in v2.0
      # compliance:  regulatory or legal truth
      # resource:    cost, time, or compute truth
    scope: global | stratum | module
    enforcement: hard | soft
      # hard = Ultra halts execution on violation
      # soft = Ultra emits a warning and logs, then continues
    verified_by: >
      How Ultra's VERIFY phase detects a violation of this axiom.
      Observable, not implementation-specific.

goals:

      statement: "..."
      priority: critical | high | medium | low
      measurable_outcome: >
        How Ultra knows this goal was achieved. Must be observable
        without implementation knowledge.
  secondary:
    - id: G-<NNN>
      statement: "..."
      measurable_outcome: "..."

# Hard limits and ethical guardrails.
# Organized by type. Ultra enforces all constraints in every planning cycle.
#
# NEW in v2.0: ethical block is a first-class constraint category (from S3).
# ═══════════════════════════════════════════════════════════════════════════════

constraints:
  global:
    # Architectural truths that bind the entire system.
    - "<hard rule that must never be violated>"
  operational:
    # Runtime limits that apply during execution.
    - "<e.g. no operation may hold a distributed lock for more than 5 seconds>"
  ethical:
    # NEW in v2.0 — value alignment and harm boundaries.
    # These are checked by Ultra's REFLECT phase.
    # Write them as observable boundaries, not vague intentions.
    - "<e.g. no user data may be used for purposes beyond what was consented to>"
    - "<e.g. automated decisions with life-affecting outcomes must have human review>"
  compliance:
    # Regulatory and legal constraints.
    - "<e.g. all PII must be encrypted at rest and in transit>"
  resource:
    # Cost, compute, time limits.
    - "<e.g. total inference cost per request must not exceed $0.01>"

# SECTION 6 · RISK MODEL
# Known failure modes. Used by Ultra's REPLAN phase.
# NEW in v2.0: blast_radius field added (from S2) — tells Ultra
# how far a failure in this module can propagate.
# ═══════════════════════════════════════════════════════════════════════════════

risk_model:
  tolerance: low | medium | high
    # The maximum risk level Ultra will accept before triggering a replan.
    # E.g. "high" = Ultra replans only on critical-risk failures.
    # "low" = Ultra replans on any high or critical risk violation.

  risks:
    - id: R-<NNN>
      description: "What could go wrong."
      severity: critical | high | medium | low
      blast_radius: system-wide | stratum-only | module-only
        # system-wide:   failure cascades to all dependent strata
        # stratum-only:  failure is contained within its stratum
        # module-only:   failure affects only this module
      mitigation: "How Ultra should respond if this risk materializes."
      affects: [<module-id>, ...]


# ═══════════════════════════════════════════════════════════════════════════════
# SECTION 7 · STRATA
# The execution core of UADF. Ordered layers executed bottom-up.
# Each stratum is a self-contained responsibility boundary.
# Ultra resolves all inter-stratum dependencies before beginning execution.
# ═══════════════════════════════════════════════════════════════════════════════

strata:
  - id: <stratum-id>                  # kebab-case, unique across the system
    name: "Stratum Name"
    order: <integer>                  # 1 = first executed; ascending = bottom-up
    intent: >
      One sentence: what conceptual responsibility this entire stratum holds.
    execution_policy:
      mode: sequential | parallel | conditional
        # sequential:  modules run in priority order, one at a time
        # parallel:    all modules start simultaneously (bonds respected)
        # conditional: stratum executes only when condition_signal is raised
      condition_signal: <signal-id>   # required when mode: conditional
      on_failure: halt | rollback | skip | retry
      retry_limit: <integer>          # max retry attempts before applying on_failure

    modules:
      - id: <module-id>               # unique across the ENTIRE system
        name: "Module Name"
        intent: >
          One sentence: the conceptual work this module performs.
        responsibility: >
          A single-responsibility declaration. If this needs two sentences,
          split the module into two. Ultra uses this to scope its micro-plan.
        priority: <integer>           # within stratum; lower number = executes first
        risk: critical | high | medium | low
        stability: volatile | stable | frozen

        obligations:
          # What this module NEEDS before it can run.
          # Name concepts, never implementations.
          - concept: "..."            # the abstract thing needed (e.g. "enriched-event")
            from: <module-id>         # which module is the source
            required: true | false    # true = hard blocker; false = optional enhancement

        contracts:
          # What this module GUARANTEES to produce when it completes.
          - concept: "..."            # the abstract thing produced
            consumers: [<module-id>, ...]  # which modules depend on this

        verify_criteria:
          # How Ultra's VERIFY phase knows this module succeeded.
          # Write as observable outcomes — no implementation language.
          - "..."

        constraints:
          # What this module must NOT do, or limits it must respect.
          - "..."

        metadata:
          tags: []
          notes: "..."                # free-form notes for human readers
          metadata: {}                # extension point


# ═══════════════════════════════════════════════════════════════════════════════
# SECTION 8 · BONDS
# The explicit dependency graph. Machine-optimized DAG projection.
# Ultra builds the execution DAG entirely from this section before running
# anything. It is a redundant but structured view of all module obligations.
#
# Bond types encode RELATIONSHIP SEMANTICS, not just topology:
#   requires:     'from' cannot start until 'to' completes successfully
#   enhances:     'from' produces better output when 'to' is available,
#                 but can run without it (quality degrades, not fails)
#   observes:     'from' reads 'to's output without depending on its
#                 availability — pure data consumption, no coupling
#   produces-for: 'from' exists solely to supply 'to' — directional feed
#
# Bond strength:
#   hard = missing dependency halts Ultra's execution plan
#   soft = missing dependency is logged; execution continues degraded
# ═══════════════════════════════════════════════════════════════════════════════

bonds:
  - from: <module-id>
    to: <module-id>
    type: requires | enhances | observes | produces-for
    strength: hard | soft
    description: "One sentence: why this bond exists."


# ═══════════════════════════════════════════════════════════════════════════════
# SECTION 9 · SIGNALS
# Named events that modules can raise and other modules can consume.
# Signals decouple cross-cutting concerns from module logic.
# Used by Ultra for: conditional stratum execution, loop termination,
# fallback activation, and cross-module state communication.
# ═══════════════════════════════════════════════════════════════════════════════

signals:
  - id: <signal-id>                   # SCREAMING-KEBAB-CASE  e.g. SIG-DRIFT-DETECTED
    name: "Signal Name"
    raised_by: [<module-id>, ...]
    consumed_by: [<module-id>, ...]
    description: "What this signal means and what it triggers."
    persistence: transient | durable
      # transient: signal is consumed once and cleared
      # durable:   signal persists across iterations until explicitly cleared

# SECTION 10 · DATA FLOWS
# NEW in v2.0 (from S2) — Conceptual data movement across strata.
# These describe WHAT moves and HOW it moves, not HOW it is implemented.
# Ultra uses flows to validate that data produced by one stratum
# can reach the strata that need it, and to plan buffer/queue sizing.
# ═══════════════════════════════════════════════════════════════════════════════

data_flows:
  - id: <flow-id>                     # kebab-case  e.g. "transaction-event-flow"
    name: "Flow Name"
    from_stratum: <stratum-id>
    to_stratum: <stratum-id>
    carries: "<abstract concept name>"  # what the flow transports
    direction: unidirectional | bidirectional | multicast
    classification: public | internal | confidential | restricted
    consistency: immediate | async | batch
    volume: stream | burst | steady
    description: "..."


# ═══════════════════════════════════════════════════════════════════════════════
# SECTION 11 · EXECUTION
# Top-level execution policy governing the entire system lifecycle.
#
# NEW in v2.0: modes block (from S2) — bootstrap, upgrade, recovery.
# These give Ultra distinct plans for first-time build, rolling updates,
# and failure recovery rather than treating all execution as identical.
# ═══════════════════════════════════════════════════════════════════════════════

execution:
  entry_stratum: <stratum-id>         # where Ultra begins
  exit_stratum: <stratum-id>          # where Ultra stops (unless replanning)

  # ── Loop policy ────────────────────────────────────────────────────────────
  # Ultra's PLAN → MICRO-PLAN → EXECUTE → VERIFY → REFLECT → REPLAN loop.
  loop_policy:
    max_iterations: <integer>
    convergence_signal: <signal-id>   # Ultra exits the loop when this is raised

  # ── Rollback policy ────────────────────────────────────────────────────────
  rollback_policy:
    enabled: true | false
    checkpoint_at: [<stratum-id>, ...]  # Ultra saves state here for rollback
    rollback_to: first-checkpoint | last-checkpoint | named
    named_checkpoint: <stratum-id>    # only if rollback_to: named

  # ── Reflection policy ──────────────────────────────────────────────────────
  # Governs Ultra's REFLECT phase — when it pauses to evaluate progress
  # and whether it should replan.
  reflection_policy:
    enabled: true | false
    reflect_after: [<stratum-id>, ...]  # strata after which Ultra reflects
    replan_threshold: critical | high | medium
      # The minimum risk severity that automatically triggers a replan.
      # "critical" = Ultra only replans for critical-severity violations.
      # "medium"   = Ultra replans for medium, high, or critical violations.
    reviewer: autonomous | human | both
      # autonomous: Ultra makes replan decisions without human input
      # human:      Ultra pauses and presents findings for human approval
      # both:       Ultra acts autonomously but notifies a human of replans

  # ── Modes ──────────────────────────────────────────────────────────────────
  # NEW in v2.0 (from S2).
  # Ultra selects the active mode based on context or explicit instruction.
  # Each mode overrides default execution behavior for its scenario.
  modes:

    bootstrap:
      # First-time initialization of the system from zero.
      description: >
        Executed when the system has no prior state. Ultra initializes
        all strata in dependency order, verifying each before proceeding.
      stratum_sequence: [<stratum-id>, ...]
        # Explicit override of default ascending order if needed.
        # If absent, Ultra uses the standard order field.
      verification_depth: minimal | standard | comprehensive
        # minimal:       check that each module started without error
        # standard:      check verify_criteria on all modules
        # comprehensive: check verify_criteria AND all axioms after every module
      on_failure: halt | partial-state-preserve
        # partial-state-preserve: keep successfully initialized strata;
        # resume from last successful checkpoint on retry

    upgrade:
      # System update while preserving availability.
      description: >
        Executed when the architecture version changes. Ultra applies
        changes to strata incrementally according to the chosen strategy.
      strategy: rolling | blue-green | canary | maintenance-window
        # rolling:             update modules one at a time; no downtime
        # blue-green:          build new version in parallel; cut over atomically
        # canary:              route a small traffic fraction to new version first
        # maintenance-window:  full downtime; simpler, riskier
      affected_strata: [<stratum-id>, ...]
        # Which strata are being upgraded. Ultra only re-executes these.
        # Unlisted strata are treated as stable and not touched.
      rollback_triggers:
        - condition: "Observable condition that signals the upgrade is failing."
          action: auto-rollback | pause-and-alert | continue-degraded
      validation_gate: >
        What Ultra must verify before cutting traffic to the new version.

    recovery:
      # Response to a failure in a running system.
      description: >
        Executed when the system has entered a degraded or failed state.
        Ultra restores service in dependency order, prioritizing the
        critical path.
      priority_strata: [<stratum-id>, ...]
        # Strata to restore first. Ordered by criticality.
        # Ultra restores these before addressing secondary strata.
      recovery_order: dependency | priority | manual
        # dependency: restore in standard bond-resolved order
        # priority:   use priority_strata list above
        # manual:     wait for human instruction on each stratum
      consistency_priority: availability | consistency | partition-tolerance
        # Which property of the CAP theorem to favor during recovery.
        # availability:          serve traffic even if data may be stale
        # consistency:           refuse requests until state is verified clean
        # partition-tolerance:   accept that some nodes may be unreachable
      degraded_operation_allowed: true | false
        # If true, Ultra may bring strata online in a reduced-capability mode
        # and progressively restore full function. If false, full recovery
        # must complete before any stratum is considered operational.

# SECTION 12 · OBSERVABILITY
# NEW in v2.0 (from S3) — What Ultra monitors and where it pauses for review.
# This section does NOT describe monitoring implementation.
# It describes WHAT signals matter and WHEN checkpoints occur.
# ═══════════════════════════════════════════════════════════════════════════════

observability:
  metrics:
    # What conceptual quantities should be tracked during execution.
    # Name the signal, not the tool.
    - concept: "..."
      signals_risk: <risk-id>         # which risk this metric detects
      threshold: "..."                # qualitative or quantitative alert point
      scope: global | stratum | module

  checkpoints:
    # Points at which Ultra formally evaluates system state.
    - id: <checkpoint-id>
      trigger: pre-execution | post-stratum | post-module | on-anomaly | milestone
      condition: "Natural-language description of when this fires."
      scope: <stratum-id> | <module-id> | global
      reviewer: autonomous | human | both
      action: approve-and-continue | modify-and-continue | abort | log-and-continue
      blocks_execution: true | false
        # true:  Ultra waits for reviewer action before proceeding
        # false: checkpoint is asynchronous; execution continues

  logging:
    level: minimal | standard | verbose
    audit_trail: true | false         # whether every decision is persisted

# ─────────────────────────────────────────────────────────────────────────────
# manifest           → identity and classification of the system
# properties         → system-wide behavioral contracts (consistency, failure mode)
# axioms             → truths that must hold; checked by VERIFY phase
# goals              → success definitions; checked by REFLECT phase
# constraints        → hard limits by type: global, operational, ethical,
#                      compliance, resource
# risk_model         → known failure modes; used by REPLAN phase
# strata             → ordered execution layers containing modules
#   modules          → atomic units of work; declare obligations + contracts
# bonds              → typed DAG of all dependencies; Ultra builds from this
# signals            → named events for conditional execution and loop control
# data_flows         → conceptual data movement across strata
# execution          → lifecycle policy: loop, rollback, reflection, modes
#   modes.bootstrap  → first-time initialization plan
#   modes.upgrade    → rolling/canary/blue-green update plan
#   modes.recovery   → failure recovery plan with consistency priority
# observability      → metrics, checkpoints, logging
# evolution          → rules governing how this spec may be modified
